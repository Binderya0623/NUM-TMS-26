  2.11 auth_service (нэвтрэх эрх)                                                                                                              
                                                            
  Admin-ын login-г үлдээнэ. sisi_id нь ихэнх тохиолдолд email эсвэл student_id. Admin-ын sisi_id-г мэдэхгүй бол дараах байдлаар email-ээр      
  ялгана:                                                   
                                                                                                                                               
  psql -U bindertsetseg -d auth_service_db <<'SQL'          
  -- Эхлээд аль user-үүд ADMIN ролтой байгааг харах
  SELECT u.id, u.sisi_id                                                                                                                       
    FROM users u                                                                                                                               
    JOIN user_roles r ON r.user_id = u.id                                                                                                      
   WHERE r.role = 'ROLE_ADMIN';                                                                                                                
                                                                                                                                               
  -- ADMIN биш бүх хэрэглэгчийг (багш, оюутан, expert) устгах                                                                                  
  DELETE FROM user_roles                                                                                                                       
   WHERE user_id IN (                                                                                                                          
     SELECT u.id FROM users u                               
     WHERE u.id NOT IN (
       SELECT user_id FROM user_roles WHERE role_name = 'ROLE_ADMIN'                                                                                
     )
   );                                                                                                                                          
                                                            
  DELETE FROM users                                                                                                                            
   WHERE id NOT IN (                                        
     SELECT user_id FROM user_roles WHERE role_name = 'ROLE_ADMIN'
   );                                                                                                                                          
  SQL
                                                                                                                                               
  Бүгдийг устгахыг хүсвэл:                                                                                                                     
  
  TRUNCATE TABLE user_roles, users RESTART IDENTITY CASCADE;
